```typescript
// admin/src/lib/api.ts
import {
  PostDto, PostCreateInput, PostUpdateInput, PostListQuery,
  AgentJobDto, AgentJobCreateInput,
} from "@artka/shared";

const API_URL = import.meta.env.VITE_API_URL ?? "/api";

class ApiError extends Error {
  constructor(public code: string, message: string, public status: number, public details?: unknown) {
    super(message);
  }
}

async function fetchJson<T>(path: string, init?: RequestInit): Promise<T> {
  const res = await fetch(`${API_URL}${path}`, {
    ...init,
    credentials: "include",
    headers: { "Content-Type": "application/json", ...init?.headers },
  });
  if (!res.ok) {
    let body: any = null;
    try { body = await res.json(); } catch {}
    throw new ApiError(
      body?.error?.code ?? "INTERNAL",
      body?.error?.message ?? `HTTP ${res.status}`,
      res.status,
      body?.error?.details,
    );
  }
  const json = (await res.json()) as { data: T };
  return json.data;
}

export const api = {
  posts: {
    list: (q: Partial<PostListQuery>) =>
      fetchJson<PostDto[]>(`/v1/admin/posts?${new URLSearchParams(q as any)}`),
    get: (id: string) => fetchJson<PostDto>(`/v1/admin/posts/${id}`),
    create: (input: PostCreateInput) =>
      fetchJson<PostDto>(`/v1/admin/posts`, { method: "POST", body: JSON.stringify(input) }),
    update: (id: string, input: Partial<PostCreateInput>) =>
      fetchJson<PostDto>(`/v1/admin/posts/${id}`, { method: "PATCH", body: JSON.stringify(input) }),
    publish: (id: string) =>
      fetchJson<PostDto>(`/v1/admin/posts/${id}/publish`, { method: "POST" }),
    unpublish: (id: string) =>
      fetchJson<{ id: string; status: string }>(`/v1/admin/posts/${id}/unpublish`, { method: "POST" }),
    archive: (id: string) =>
      fetchJson<{ id: string; status: string }>(`/v1/admin/posts/${id}/archive`, { method: "POST" }),
    render: (id: string) =>
      fetchJson<PostDto>(`/v1/admin/posts/${id}/render`, { method: "POST" }),
    revisions: (id: string) =>
      fetchJson<any[]>(`/v1/admin/posts/${id}/revisions`),
    restoreRevision: (id: string, revId: number) =>
      fetchJson<{ ok: true }>(`/v1/admin/posts/${id}/revisions/${revId}/restore`, { method: "POST" }),
    renderPreview: (body_md: string, lang: "ru" | "en") =>
      fetchJson<{ body_html: string; toc: any[] }>(`/v1/admin/posts/render/preview`, {
        method: "POST",
        body: JSON.stringify({ body_md, lang }),
      }),
  },
  jobs: {
    list: (q: { status?: string; kind?: string; limit?: number; offset?: number } = {}) =>
      fetchJson<AgentJobDto[]>(`/v1/admin/jobs?${new URLSearchParams(q as any)}`),
    get: (id: string) =>
      fetchJson<{ job: AgentJobDto; runs: any[]; artifacts: any[] }>(`/v1/admin/jobs/${id}`),
    create: (input: AgentJobCreateInput) =>
      fetchJson<AgentJobDto>(`/v1/admin/jobs`, { method: "POST", body: JSON.stringify(input) }),
    cancel: (id: string) =>
      fetchJson<{ id: string; cancelled: true }>(`/v1/admin/jobs/${id}/cancel`, { method: "POST" }),
    retry: (id: string) =>
      fetchJson<AgentJobDto>(`/v1/admin/jobs/${id}/retry`, { method: "POST" }),
  },
  media: {
    list: () => fetchJson<any[]>(`/v1/admin/media`),
    upload: async (file: File): Promise<any> => {
      const fd = new FormData();
      fd.append("file", file);
      const res = await fetch(`${API_URL}/v1/admin/media/upload`, {
        method: "POST", body: fd, credentials: "include",
      });
      if (!res.ok) throw new Error(`Upload failed ${res.status}`);
      return ((await res.json()) as { data: any }).data;
    },
    delete: (id: number) =>
      fetchJson<{ ok: true }>(`/v1/admin/media/${id}`, { method: "DELETE" }),
  },
```

```typescript
// admin/src/lib/auth.ts
import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";
import { useNavigate } from "@tanstack/react-router";

const API_URL = import.meta.env.VITE_API_URL ?? "/api";

interface Session {
  user: { id: string; email: string; name: string; role: string };
  session: { id: string; userId: string; expiresAt: string };
}

async function getSession(): Promise<Session | null> {
  const res = await fetch(`${API_URL}/auth/get-session`, { credentials: "include" });
  if (!res.ok) return null;
  const json = await res.json();
  return json && json.user ? json : null;
}

export function useSession() {
  return useQuery({
    queryKey: ["session"],
    queryFn: getSession,
    staleTime: 60_000,
    refetchOnWindowFocus: false,
  });
}

export function useSignIn() {
  const qc = useQueryClient();
  const navigate = useNavigate();
  return useMutation({
    mutationFn: async (input: { email: string; password: string }) => {
      const res = await fetch(`${API_URL}/auth/sign-in/email`, {
        method: "POST",
        credentials: "include",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(input),
      });
      if (!res.ok) {
        const body = await res.json().catch(() => ({}));
        throw new Error(body?.message ?? "Sign in failed");
      }
      return res.json();
    },
    onSuccess: () => {
      qc.invalidateQueries({ queryKey: ["session"] });
      navigate({ to: "/" });
    },
  });
}

export function useSignOut() {
  const qc = useQueryClient();
  const navigate = useNavigate();
  return useMutation({
    mutationFn: async () => {
      await fetch(`${API_URL}/auth/sign-out`, { method: "POST", credentials: "include" });
    },
    onSuccess: () => {
      qc.setQueryData(["session"], null);
      navigate({ to: "/login" });
    },
  });
}
```

```typescript
export function formatDate(d: Date | string): string {
  const date = typeof d === "string" ? new Date(d) : d;
  return new Intl.DateTimeFormat("ru-RU", {
    year: "numeric", month: "2-digit", day: "2-digit",
    hour: "2-digit", minute: "2-digit",
  }).format(date);
}

export function formatCost(usd: string | number): string {
  const n = typeof usd === "string" ? parseFloat(usd) : usd;
  if (n < 0.01) return `$${(n * 100).toFixed(2)}¢`;
  return `$${n.toFixed(4)}`;
}

export function formatDuration(ms: number): string {
  if (ms < 1000) return `${ms}ms`;
  if (ms < 60_000) return `${(ms / 1000).toFixed(1)}s`;
  return `${(ms / 60_000).toFixed(1)}m`;
}
```

```tsx
import { Navigate } from "@tanstack/react-router";
import { useSession } from "@/lib/auth";

export function RequireAuth({ children }: { children: React.ReactNode }) {
  const { data: session, isPending } = useSession();
  if (isPending) {
    return <div className="p-8 text-muted-foreground">Loading…</div>;
  }
  if (!session) {
    return <Navigate to="/login" />;
  }
  if (session.user.role !== "admin" && session.user.role !== "editor") {
    return <div className="p-8 text-destructive">Access denied. Role: {session.user.role}</div>;
  }
  return <>{children}</>;
}
```

```typescript
import { useEffect, useState } from "react";

export function useDebouncedValue<T>(value: T, delay = 500): T {
  const [debounced, setDebounced] = useState(value);
  useEffect(() => {
    const id = setTimeout(() => setDebounced(value), delay);
    return () => clearTimeout(id);
  }, [value, delay]);
  return debounced;
}
```

```tsx
import { useForm } from "react-hook-form";
import { zodResolver } from "@hookform/resolvers/zod";
import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";
import { useNavigate } from "@tanstack/react-router";
import { PostCreateInput, type PostDto } from "@artka/shared";
import { api } from "@/lib/api";
import { Button } from "@/components/ui/button";
import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";
import { FrontmatterForm } from "./FrontmatterForm";
import { PreviewIframe } from "./PreviewIframe";
import { RevisionsList } from "./RevisionsList";
import { useDebouncedValue } from "@/lib/hooks/use-debounced-value";

interface Props {
  initial?: PostDto;
  onSaved?: (post: PostDto) => void;
}

export function PostEditor({ initial, onSaved }: Props) {
  const qc = useQueryClient();
  const navigate = useNavigate();

  const form = useForm<PostCreateInput>({
    resolver: zodResolver(PostCreateInput),
    defaultValues: initial ? mapToInput(initial) : {
      slug: "", lang: "ru", kind: "post", status: "draft",
      title: "", description: "", body_md: "",
      pub_date: new Date(), keywords: [], tags: [], extra: {},
      author: "Артём", display_order: 0, pinned: false, manually_edited: false,
    } as any,
  });

  const [body, setBody] = useState<string>(initial?.body_md ?? "");
  const debouncedBody = useDebouncedValue(body, 500);

  // Sync body_md into form when CodeMirror changes
  useEffect(() => {
    form.setValue("body_md", body);
  }, [body, form]);

  const preview = useQuery({
    queryKey: ["render-preview", debouncedBody, form.watch("lang")],
    queryFn: () => api.posts.renderPreview(debouncedBody, form.watch("lang")),
    enabled: debouncedBody.length > 0,
  });

  const create = useMutation({
    mutationFn: api.posts.create,
    onSuccess: (post) => {
      qc.invalidateQueries({ queryKey: ["posts"] });
      onSaved?.(post);
      if (!initial) navigate({ to: `/posts/${post.id}` });
    },
  });
  const update = useMutation({
    mutationFn: (input: Partial<PostCreateInput>) => api.posts.update(initial!.id, input),
    onSuccess: (post) => {
      qc.invalidateQueries({ queryKey: ["posts"] });
      onSaved?.(post);
    },
  });
  const publish = useMutation({
    mutationFn: () => api.posts.publish(initial!.id),
    onSuccess: () => qc.invalidateQueries({ queryKey: ["posts"] }),
  });

  const onSubmit = form.handleSubmit((data) => {
    if (initial) update.mutate(data);
    else create.mutate(data);
  });

  return (
    <div className="grid grid-cols-2 gap-4 h-[calc(100vh-12rem)]">
      <div className="flex flex-col gap-3 overflow-y-auto pr-3">
        <div className="flex justify-between items-center">
          <h2 className="text-lg font-semibold">{initial ? `Edit: ${initial.title}` : "New post"}</h2>
          <div className="flex gap-2">
            <Button onClick={onSubmit} disabled={create.isPending || update.isPending}>
              Save
            </Button>
            {initial && (
              <Button onClick={() => publish.mutate()} variant="default" disabled={publish.isPending}>
                Publish
              </Button>
            )}
          </div>
        </div>
        <Tabs defaultValue="meta">
          <TabsList>
            <TabsTrigger value="meta">Frontmatter</TabsTrigger>
            <TabsTrigger value="body">Body</TabsTrigger>
            {initial && <TabsTrigger value="revisions">Revisions</TabsTrigger>}
          </TabsList>
          <TabsContent value="meta"><FrontmatterForm form={form} /></TabsContent>
          <TabsContent value="body">
            <CodeMirror
              value={body}
              onChange={setBody}
              extensions={[markdown()]}
              height="500px"
              basicSetup={{ lineNumbers: true, foldGutter: true }}
            />
          </TabsContent>
          {initial && (
            <TabsContent value="revisions">
              <RevisionsList postId={initial.id} />
            </TabsContent>
          )}
        </Tabs>
      </div>
      <div className="border rounded overflow-hidden">
        <PreviewIframe html={preview.data?.body_html ?? "<p style=\"color:#888\">Type to preview…</p>"} />
      </div>
    </div>
  );
}

function mapToInput(p: PostDto): PostCreateInput {
  return {
    slug: p.slug, lang: p.lang, kind: p.kind, status: p.status,
    title: p.title, description: p.description, summary: p.summary ?? undefined,
    keywords: p.keywords, faq: p.faq ?? undefined, tags: p.tags,
    cover: p.cover ?? undefined, cover_alt: p.cover_alt ?? undefined,
    author: p.author, pub_date: p.pub_date, updated_date: p.updated_date ?? undefined,
    extra: p.extra, body_md: p.body_md,
    source_hash: p.source_hash ?? undefined, manually_edited: p.manually_edited,
    display_order: p.display_order, pinned: p.pinned,
  };
}
```

```typescript
import { useEffect } from "react";
import { useQueryClient } from "@tanstack/react-query";

const API_URL = import.meta.env.VITE_API_URL ?? "/api";

export function useJobsStream() {
  const qc = useQueryClient();
  useEffect(() => {
    const lastId = localStorage.getItem("jobs-stream-last-id");
    const es = new EventSource(`${API_URL}/v1/admin/jobs/stream`, { withCredentials: true });

    const onEvent = (e: MessageEvent) => {
      if (e.lastEventId) localStorage.setItem("jobs-stream-last-id", e.lastEventId);
      qc.invalidateQueries({ queryKey: ["jobs"] });
      try {
        const data = JSON.parse(e.data);
        if (data.job_id) {
          qc.invalidateQueries({ queryKey: ["job", data.job_id] });
        }
      } catch {}
    };

    es.addEventListener("job.created", onEvent);
    es.addEventListener("job.updated", onEvent);
    es.addEventListener("run.started", onEvent);
    es.addEventListener("run.tokens", onEvent);
    es.addEventListener("run.finished", onEvent);

    es.onerror = () => {
      // EventSource auto-reconnects; Last-Event-ID set by browser using id field
    };

    return () => es.close();
  }, [qc]);
}
```

```tsx
import { useState } from "react";
import { useMutation } from "@tanstack/react-query";
import { useNavigate } from "@tanstack/react-router";
import { api } from "@/lib/api";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Button } from "@/components/ui/button";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
import type { AgentKind } from "@artka/shared";

const KINDS: AgentKind[] = ["draft_from_url", "translate", "social_drafts", "rss_monitor", "daily_digest", "rerender_all", "export_to_git"];

export function AgentTriggerWizard({ initialKind }: { initialKind?: AgentKind }) {
  const navigate = useNavigate();
  const [kind, setKind] = useState<AgentKind>(initialKind ?? "draft_from_url");
  const [payload, setPayload] = useState<Record<string, any>>({});

  const create = useMutation({
    mutationFn: () => api.jobs.create({ kind, payload } as any),
    onSuccess: (job) => navigate({ to: `/jobs/${job.id}` }),
  });

  return (
    <div className="max-w-md space-y-3">
      <div>
        <Label>Agent kind</Label>
        <Select value={kind} onValueChange={(v) => { setKind(v as AgentKind); setPayload({}); }}>
          <SelectTrigger><SelectValue /></SelectTrigger>
          <SelectContent>
            {KINDS.map((k) => <SelectItem key={k} value={k}>{k}</SelectItem>)}
          </SelectContent>
        </Select>
      </div>

      {kind === "draft_from_url" && (
        <div>
          <Label>Source URL</Label>
          <Input
            type="url"
            value={payload.url ?? ""}
            onChange={(e) => setPayload({ ...payload, url: e.target.value, target_lang: "ru" })}
          />
        </div>
      )}
      {kind === "translate" && (
        <div>
          <Label>Post ID (UUID)</Label>
          <Input
            value={payload.post_id ?? ""}
            onChange={(e) => setPayload({ ...payload, post_id: e.target.value, force: false })}
          />
        </div>
      )}
      {kind === "social_drafts" && (
        <div>
          <Label>Post ID (UUID)</Label>
          <Input
            value={payload.post_id ?? ""}
            onChange={(e) => setPayload({ ...payload, post_id: e.target.value })}
          />
        </div>
      )}
      {kind === "rss_monitor" && (
        <div className="text-sm text-muted-foreground">
          No payload — uses default feeds from agents config.
        </div>
      )}
      {kind === "daily_digest" && (
        <div>
          <Label>Since hours</Label>
          <Input
            type="number"
            value={payload.since_hours ?? 24}
            onChange={(e) => setPayload({ ...payload, since_hours: Number(e.target.value) })}
          />
        </div>
      )}
      {kind === "rerender_all" && (
        <div>
          <Label>Reason</Label>
          <Input
            value={payload.reason ?? ""}
            onChange={(e) => setPayload({ ...payload, reason: e.target.value })}
          />
        </div>
      )}
      {kind === "export_to_git" && (
        <div>
          <Label>Post ID</Label>
          <Input
            value={payload.post_id ?? ""}
            onChange={(e) => setPayload({ ...payload, post_id: e.target.value })}
          />
        </div>
      )}

      <Button onClick={() => create.mutate()} disabled={create.isPending}>
        {create.isPending ? "Triggering…" : "Trigger"}
      </Button>
      {create.isError && <p className="text-sm text-destructive">{(create.error as Error).message}</p>}
    </div>
  );
}
```
